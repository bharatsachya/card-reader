# Pinned so `pip install -r requirements.txt` reproduces this exact environment.
# Versions are what was resolved on Python 3.13; all support 3.11+.

# --- Web framework + ASGI server ---
fastapi==0.141.1
uvicorn[standard]==0.53.0
# Required by FastAPI to parse multipart/form-data, i.e. file uploads.
# FastAPI raises a RuntimeError on the first UploadFile route without it.
python-multipart==0.0.32

# --- HTTP client for the model's OpenAI-compatible endpoint ---
httpx==0.28.1

# --- Image preprocessing: EXIF rotation, resize, re-encode ---
pillow==12.3.0
# HEIC/HEIF decoding. Not optional: HEIC is the default iPhone camera format,
# and Pillow cannot read it on its own.
pillow-heif==1.7.0

# --- .xlsx generation ---
openpyxl==3.1.5

# --- Deterministic phone parsing / E164 formatting ---
phonenumbers==9.0.39

# --- Card detection (perspective crop before resize) ---
# HEADLESS, not plain opencv-python: the full package links GUI toolkits (X11,
# GTK, Qt) that do not exist in a slim container and would not be used if they
# did. Same cv2 module, ~35 MB smaller, no system packages to install.
# numpy comes with it and is used only to pass pixel buffers between the two.
opencv-python-headless==4.12.0.88

# --- Persistence ---
# Async SQLite driver. Needed because LeadStore's methods are async: the stdlib
# `sqlite3` module is blocking, so calling it directly from a coroutine would
# stall the entire event loop -- including every other card's model call and
# every poll in flight -- for the duration of each write. aiosqlite runs the
# same sqlite3 calls on a worker thread and awaits the result.
aiosqlite==0.22.1

# --- Authentication: verifying Clerk session tokens ---
# PyJWT does the JWT work; [crypto] pulls in `cryptography` for the RS256
# signature check. Only PUBLIC keys are used -- see app/auth.py.
PyJWT[crypto]==2.14.0
# Mozilla's CA bundle. Imported directly because Python does not use the OS
# certificate store, so JWKS fetching fails without it on macOS and on slim
# Linux images.
certifi==2026.7.22
